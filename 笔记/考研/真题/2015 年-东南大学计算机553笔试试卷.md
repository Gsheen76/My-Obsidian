#### 1. 求$e^x$的值
要求：$|\frac{x^n}{n!}-\frac{x^{n-1}}{n-1!}|>10^{-7}$ ，已知： $e^x = 1 + \frac{x^1}{1!} + \frac{x^2}{2!} + \frac{x^3}{3!} + \cdots$
题意推测：根据级数展开计算 $e^x$，当前后两项之差小于某阈值时终止， 输出计算结果。
```c++
#include<bits/stdc++.h>
using namespace std;

int main(){
	double x,ans=1.0,temp=1.0;
	cin>>x;
	for(int i=1;;i++){
		double prev=temp;
		temp=temp*x/i;
		ans+=temp;
		if(fabs(temp-prev)<=1e-7)
			break;
	}
	cout<<ans<<endl;
	return 0;
}
```

#### 2. 编写一个递归函数 
功能：输入一个字符串，字符与字符之间含空格分隔，输出一个整型值。 
例如：输入 1 x 2 y z 3 d h 4 g 5，输出 54321。 
```c++
#include<iostream>
#include<string>
using namespace std;

int main()
{
    string s;
    getline(cin, s);
    int n=s.size();
    int ans=0,cnt=1;
    for(int i=0;i<n;i++)
    {
        if(s[i]>='0' && s[i]<='9')
        {
            ans+=cnt*(s[i]-'0');
            cnt*=10;
        }
    }
    cout<<ans;
    return 0;
}
```

#### 3. 使用随机数函数 srand() 和 time() 生成随机数来模拟掷骰子 
(1) 分别掷两次，将两次掷出的值相加，存入二维数组 A，数组的“行”表示第一次投
掷结果，数组的“列”表示第二次投掷结果； （可能回忆不，，推测数组元素是记
录该行列组合和值。） 
(2) 如果投掷 1000 次，将它们的和的值及其出现的次数写入另一二维数组 B。编写
函数完成以上功能。（本题确为原话，题意本身略模糊，就是比较抽象。） 
```c++
#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int main()
{
    srand(time(0));

    int A[6][6];     // 6×6
    int B[11][2] = {0};  // 和 2~12 → 11种

    // ===== 构造 A =====
    for(int i = 0; i < 6; i++){
        for(int j = 0; j < 6; j++){
            A[i][j] = (i + 1) + (j + 1);
        }
    }

    // ===== 初始化 B =====
    for(int i = 0; i < 11; i++){
        B[i][0] = i + 2;  // 和
        B[i][1] = 0;      // 次数
    }

    // ===== 模拟 1000 次 =====
    for(int k = 0; k < 1000; k++){
        int d1 = rand() % 6 + 1;
        int d2 = rand() % 6 + 1;

        int sum = d1 + d2;

        B[sum - 2][1]++;  // 统计次数
    }

    return 0;
}	
```

#### 4. 编写 FindRepStr() 函数
此函数功能为：在字符串 str 中查找目标字符串 findStr，并用 replaceStr 来替换，最后输出替换后的 str。函数原型为： void FindRepStr(char str[], const char findStr[], const char replaceStr[]); 
```c++
#include <iostream>
#include <cstring>
using namespace std;

void FindrRepStr(char str[], const char findStr[], const char replaceStr[]) {
    char temp[200];
    int i = 0;

    while (str[i] != '\0') {
        int j = 0;
        int k = i;

        // 尝试匹配（不改变 i）
        while (str[k] == findStr[j] && findStr[j] != '\0') {
            k++;
            j++;
        }

        if (findStr[j] == '\0') {  // 找到匹配
            int pos = 0;

            // 复制前半部分
            for (int m = 0; m < i; m++)
                temp[pos++] = str[m];

            // 插入 replaceStr
            for (int m = 0; replaceStr[m] != '\0'; m++)
                temp[pos++] = replaceStr[m];

            // 复制后半部分
            for (int m = k; str[m] != '\0'; m++)
                temp[pos++] = str[m];

            temp[pos] = '\0';

            strcpy(str, temp);

            i = 0;
        } 
        else
            i++;  // 没匹配就继续
    }
}

int main() {
    char str[100], findStr[100], replaceStr[100];

    cout << "请输入字符串：";
    cin.getline(str, 100);

    cout << "请输入要查找的字符串：";
    cin.getline(findStr, 100);

    cout << "请输入要替换的字符串：";
    cin.getline(replaceStr, 100);

    FindrRepStr(str, findStr, replaceStr);

    cout << "替换后的字符串为：" << str << endl;

    return 0;
}
```

#### 5. 编写一个 Teacher 类 
要求：类含有数据成员：教师编号、姓名、性别、出生年月、入职年月，以及如下成员
函数： 
(1) 带有默认参数的构造函数（默认出生年月为 1900-01-01），同时需提供复制构造
函数。
(2) 2016 年进行新一轮聘用，男教师满 60 岁、女教师满 55 岁，则到达退休年龄，
编写函数输出应退休的教师姓名和编号。 
(3) 如果满退休年龄的女教师入职时间未满 35 年，则继续聘用，编写函数输出应继
续聘用的满退休年龄女教师姓名和编号。 
(4) 编写 main 函数测试该类，创建若干教师对象，设定出生年月、入职年月、性别等
信息，验证上述逻辑。
```c++
#include<iostream>
using namespace std;

class Teacher {
private:
    int teacherID;
    string name;
    string gender;
    string birthdate;
    string joindate;
public:
    Teacher(): teacherID(0), name(""), gender(""), birthdate("1900-01-01"), joindate("") {}
    Teacher(int id, string n, string g, string b, string j): teacherID(id), name(n), gender(g), birthdate(b), joindate(j) {}
    Teacher(Teacher& t) {
        teacherID = t.teacherID;
        name = t.name;
        gender = t.gender;
        birthdate = t.birthdate;
        joindate = t.joindate;  
    }
    int age(){//2016
        int birthYear = stoi(birthdate.substr(0, 4));
        int age = 2016 - birthYear;
        return age;
    }
    int joinYears() {
        int joinYear = stoi(joindate.substr(0, 4));
        int years = 2016 - joinYear;
        return years;
    }
    void retire() {
        if(gender=="男"&&age()>=60||gender=="女"&&age()>=55) {
            cout << name << " " << teacherID << " 已退休。" << endl;
        } else {
            cout << name << " " << teacherID << " 未退休。" << endl;
        }
    }
    void retire_join(){
        if(gender=="女"&&age()>=55&&joinYears()<35){
            cout << name << " " << teacherID << endl;
        }
    }
};

int main(){
    Teacher t1(1, "张三", "男", "1950-05-20", "1980-09-01");
    Teacher t2(2, "李四", "女", "1960-10-15", "1985-03-01");
    Teacher t3(3, "王五", "男", "1955-12-30", "1990-01-01");
    Teacher t4(4, "赵六", "女", "1965-07-10", "1995-06-01");
    t1.retire();
    t2.retire();    
    t3.retire();
    t4.retire();
    cout << "退休但工龄不足35年的女教师：" << endl;
    t1.retire_join();
    t2.retire_join();
    t3.retire_join();
    t4.retire_join();
    return 0;
}
```
