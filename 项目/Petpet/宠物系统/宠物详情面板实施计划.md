        self.assertTrue(snap["active"])
        self.assertEqual(snap["level"], 1)
        self.assertEqual(snap["xp_next"], 100)
        self.assertEqual(snap["affection_next"], 30)
        self.assertEqual(len(snap["outfits"]), 2)
        self.assertFalse(any(o["equipped"] for o in snap["outfits"]))
        self.assertTrue(snap["avatar_path"])

    def test_unowned_ice_cream_snapshot(self):
        s = _fresh_state()
        snap = pet_profile_snapshot(s, "ice_cream")
        self.assertFalse(snap["owned"])
        self.assertFalse(snap["active"])
        self.assertEqual(snap["price"], 760)
        self.assertEqual(snap["outfits"], [])

    def test_owned_active_pet_reads_facade_and_outfit_flags(self):
        s = _fresh_state()
        s["owned_pet_ids"] = ["lunch_meat", "ice_cream"]
        s["owned_outfits"] = ["dinosaur_suit"]
        s["equipped_outfit"] = "dinosaur_suit"
        s["pets"]["ice_cream"]["owned_outfits"] = []  # 非当前宠物读自身 profile
        ice = pet_profile_snapshot(s, "ice_cream")
        self.assertTrue(ice["owned"])
        lunch = pet_profile_snapshot(s, "lunch_meat")
        dino = next(o for o in lunch["outfits"] if o["id"] == "dinosaur_suit")
        self.assertTrue(dino["owned"] and dino["equipped"])

    def test_active_pet_name_falls_back_to_facade(self):
        s = _fresh_state()
        s["pet_name"] = "烟花"
        snap = pet_profile_snapshot(s, "lunch_meat")
        self.assertEqual(snap["name"], "烟花")

```python
"""宠物详情面板：快照数据与窗口。"""

from __future__ import annotations

from petpet.app import pets as pet_registry
from petpet.progression import core as progression

# 与 pet.py:468 xp_to_next 保持同一曲线（包内不 import 根模块）。
_XP_TO_NEXT = lambda level: int(100 * (level ** 1.5))  # noqa: E731


def pet_profile_snapshot(state: dict, pet_id: str) -> dict:
    """汇总一只宠物的展示数据；非当前宠物读 state["pets"][pet_id]。"""
    definition = pet_registry.pet_definition(pet_id)
    active_id = state.get("active_pet_id", pet_registry.DEFAULT_PET_ID)
    profile = state if pet_id == active_id else (
        state.get("pets", {}).get(pet_id, {})
    )
    owned = pet_id == active_id or pet_id in (
        state.get("owned_pet_ids") or ()
    )

    name = profile.get("name") or profile.get("pet_name")
    if pet_id == active_id:
        name = name or state.get("pet_name") or state.get("name")
    if not (isinstance(name, str) and name.strip()):
        name = definition.get("default_name", pet_id)

    affection_level = int(profile.get("affection_level") or 1)
    affection_points = int(profile.get("affection_points") or 0)
    level = int(profile.get("level") or 1)
    owned_outfits = profile.get("owned_outfits") or []
    equipped = profile.get("equipped_outfit")

    outfits = []
    for outfit_id, outfit in progression.OUTFIT_DEFINITIONS.items():
        if outfit.get("pet_id") != pet_id:
            continue
        outfits.append({
            "id": outfit_id,
            "name": outfit.get("name", outfit_id),
            "icon": outfit.get("icon", ""),
            "price": int(outfit.get("price", 0)),
            "description": outfit.get("description", ""),
            "owned": outfit_id in owned_outfits,
            "equipped": equipped == outfit_id,
        })

    return {
        "id": pet_id,
        "name": name,
        "default_name": definition.get("default_name", pet_id),
        "description": definition.get("description", ""),
        "price": int(definition.get("price", 0) or 0),
        "owned": owned,
        "active": pet_id == active_id,
        "level": level,
        "xp": int(profile.get("xp") or 0),
        "xp_next": _XP_TO_NEXT(level),
        "affection_level": affection_level,
        "affection_points": affection_points,
        "affection_next": progression.affection_to_next(
            {"affection_level": affection_level}
        ),
        "avatar_path": pet_registry.pet_avatar_path(pet_id),
        "preview_path": pet_registry.pet_asset_path(pet_id, "desktop", "idle"),
        "outfits": outfits,
    }
```
