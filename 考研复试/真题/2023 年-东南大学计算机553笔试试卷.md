```C++
int x = 6;
```

### 2. 写出下面程序输出结果 
```c++
int main(
	const char* a[5] = {"orange","purple", "red", "yellow", "green"};
	const char* p1=a[0], * p2= a[0];
	for (int i=0;i<5;i++) {
		if (strcmp(p1, a[i]) < 0) {
			p1 = a[i];
		}
		else if (strcmp(p2, a[i]) > 0) {
			p2 = a[i];
		}
		cout << p1 << "," << p2 << endl;
	}
	return 0;
}
```
- **输出：**
```text
orange,orange
purple,orange
red,orange
yellow,orange
yellow,green
```

### 3. 写出下面程序输出结果 
```c++
class A {
private:
    int n;
public:
    A(int a = 0) {
        n = a;
        cout << "A Constructor " << n << endl;
    }
    A(const A& a) {
        n = a.n;
        cout << "A copy constructor " << n << endl;
    }
    virtual void print() {
        cout << "A n=" << n << endl;
    }
    ~A() {
        cout << "A Destructor " << n << endl;
    }
};

void fun(A a) {
    static A b(a);
}

int main() {
    A a(10);
    A a1(a);
    A a2(5);
    fun(a2);
    fun(a2);
    return 0;
}
```
- **输出：**
```text
A Constructor 10        // ① a(10)
A copy constructor 10   // ② a1(a)
A Constructor 5         // ③ a2(5)
A copy constructor 5    // ④ fun(a2) 传参
A copy constructor 5    // ④ static A b(a) 第一次初始化
A Destructor 5          // ④ 形参销毁
A copy constructor 5    // ⑤ fun(a2) 传参
A Destructor 5          // ⑤ 形参销毁
A Destructor 5          // ⑥ 析构 a2
A Destructor 10         // ⑥ 析构 a1
A Destructor 10         // ⑥ 析构 a
A Destructor 5          // ⑥ 静态变量 b 析构
```

### 4. 写出下面程序输出结果 
```c++
class person {
public:
    string name;
    person(string n="per") {
	    name = n;
    }
    virtual void print() {
	    cout << "name:" << name << endl;
    }
};

class student : public person{
public:
    string studentName;
    student(string stu="stu") {
	    studentName = stu;
    }
    void print() {
	    person::print();
	    cout << "studentName:" << studentName << endl;
    }
};

class perStudent : public student {
public:
    string perS;
public:
    perStudent(string p="pers") {
	    perS = p;
    }
    void print() {
	    student::print();
	    cout << "perS:" << perS << endl;
    }
};

int main() {
    person p("p1");
    student stu("s1");
    perStudent per("per1");
    person* pr = &p;
    pr->print();
    pr = &stu;
    pr->print();
    pr = &per;
    pr->print();
    return 0;
}
```
- **输出：**
```text
name:p1
```
