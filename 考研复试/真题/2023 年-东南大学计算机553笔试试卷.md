### 1. 写出下面程序输出结果 
```C++
int x = 6;
int main() {
	int x = 0;
	cout << x++ << end1;
	cout <<:: x << endl;
	cout << x << end1;
	if (x/2) {
		cout << "true" << end1;
	else {
		cout << "true" << end1;
	}
	return 0;
}
```
- **输出：**
```c++
0
6
1
true
```

### 2. 写出下面程序输出结果 
```c++
int main(
	const char* a[5] = {"orange","purple", "red", "yellow", "green"};
	const char* p1=a[0], * p2= a[0];
	for (int i=0;i<5;i++) {
		if (strcmp(p1, a[i]) < 0) {
			p1 = a[i];
		}
		else if (strcmp(p2, a[i]) > 0) {
			p2 = a[i];
		}
		cout << p1 << "," << p2 << endl;
	}
	return 0;
}
```
- **输出：**
```c++
orange,orange
purple,orange
red,orange
yellow,orange
yellow,green
```

### 3. 写出下面程序输出结果 
```c++
class A {
private:
    int n;
public:
    A(int a = 0) {
        n = a;
        cout << "A Constructor " << n << endl;
    }
    A(const A& a) {
        n = a.n;
        cout << "A copy constructor " << n << endl;
    }
    virtual void print() {
        cout << "A n=" << n << endl;
    }
    ~A() {
        cout << "A Destructor " << n << endl;
    }
};

void fun(A a) {
    static A b(a);
}

int main() {
    A a(10);
    A a1(a);
    A a2(5);
    fun(a2);
    fun(a2);
    return 0;
}
```
- **输出：**
```c++
A Constructor 10
A copy constructor 10
A Constructor 5
A copy constructor 5
A copy constructor 5
```
