### 1. 对给定的字符串 b，使用 c 来代替 b 中出现的 a 子串。
比如 b 为 I love nanjing, I love China, a 为 love, c 为 like，则替换之后为 I like nanjing, I like China。函数原型为 `void replace(char* b, const char* a, const char* c)`（不可以使用标准模板）。
```c++
#include <iostream>
#include <cstring>
using namespace std;

void replace(char* b, const char* a, const char* c) {
    int lenB = strlen(b);
    int lenA = strlen(a);
    int lenC = strlen(c);
    
    // 统计需要替换的次数
    int count = 0;
    char* pos = b;
    while ((pos = strstr(pos, a)) != NULL) {
        count++;
        pos += lenA;
    }
    
    int newLen = lenB + count * (lenC - lenA);
    char* temp = new char[newLen + 1];
    
    int i = 0;  // b 的索引
    int k = 0;  // temp 的索引
    
    while (b[i] != '\0') {
        bool match = true;
        for (int j = 0; j < lenA; j++) { // 检查当前位置是否匹配子串 a
            if (b[i + j] != a[j]) {
                match = false;
                break;
            }
        }
        
        if (match) {
            // 匹配成功，替换为 c
            for (int j = 0; j < lenC; j++) {
                temp[k++] = c[j];
            }
            i += lenA;  // 跳过 a 子串
        } else {
            temp[k++] = b[i++];
        }
    }
    temp[k] = '\0';   
    strcpy(b, temp);
    delete[] temp;
}
```

### 2. 2. 随机生成两组 1-13 的数字，一组 13 个元素（是扑克牌），要将其按照从小到大的顺序排序，并且统计出现次数分别为 2,3,4 的牌的种类数目
比如：
pairs1: A, A, 2, 2, 4, 4, 4, 6, 7, 10, J, Q, K  
twos:2   threes: 1   fours: 0
pairs2: A, 2, 3, 4, 4, 4, 4, 6, 7, 10, 10, Q, K  
twos:1   threes: 0   fours: 1
```c++
#include<iostream>
#include<ctime>
#include<algorithm>
using namespace std;

int main(){
    srand(time(0));
    int pairs[2][13];
    int twos[2]={0};
    int threes[2]={0};
    int fours[2]={0};
    for(int i=0;i<2;i++){
        for(int j=0;j<13;j++){
            pairs[i][j]=rand()%13+1;
        }
    }
    sort(pairs[0],pairs[0]+13);
    sort(pairs[1],pairs[1]+13);
    for(int i=0;i<2;i++){
        int count=1;
        for(int j=0;j<13;j++){
            if(j<12&&pairs[i][j]==pairs[i][j+1]){
                count++;
            }
            else{
                if(count==2){
                    twos[i]++;
                }
                else if(count==3){
                    threes[i]++;
                }
                else if(count==4){
                    fours[i]++;
                }
                count=1;
            }
        }
    }
    cout<<"twos: "<<twos[0]<<" threes: "<<threes[0]<<" fours: "<<fours[0]<<endl;
    cout<<"twos: "<<twos[1]<<" threes: "<<threes[1]<<" fours: "<<fours[1]<<endl;
    return 0;
}
```

### 3. 定义一个抽象 student 类和两个派生类，分别为 SStudent、CSStudent。  
Student 类中有一个名字数据成员，一个构造函数
SStudent 有 1 个数据成员：fee，一个构造函数
CSStudent 有两个数据成员：lowGradeFee、highGradeFee，一个构造函数
要可以计算总学费与输出名字（如果有的话）  
SStudent 费用计算：fee\*4  
CSStudent 费用计算：lowGradeFee\*0.5+highGradeFee\*0.5
```c++
#include<iostream>
using namespace std;

class Student{
private:
    string name;
public:
    Student(string n):name(n){}
    virtual int calculateFee()=0;
};

class SStudent:public Student{
private:
    int fee;
public:
    SStudent(string n,int f){
        Student(n);
        fee=f;
    }
    int calculateFee() override{
        return fee*4;
    }
};

class CSStudent:public Student{
private:
    int lowGradeFee;
    int highGradeFee;
public:
    CSStudent(string n,int l,int h){
        Student(n);
        lowGradeFee=l;
        highGradeFee=h;
    }
    int calculateFee() override{
        return lowGradeFee*0.5 + highGradeFee*0.5;
    }
};

int main(){
    SStudent s("John",1000);    
    CSStudent cs("Alice",500,1500);

    cout<<"Fee for "<<cs.calculateFee()<<endl;
```

### 4. 设计一个 MySet，是一个集合类，其中有 2 个数据成员，为整形指针 ptr，表示大小的整数 size，并且此数组中的成员不可以重复。
要求实现以下功能：
(1) 有构造函数、析构函数、复制构造函数  
(2) 可以实现插入一个元素，删除一个元素  
(3) 重载运算符+和<<  
(4) 可以查找一个元素，返回下标
```c++
class Myset{
private:
	int *ptr;
	int size;
public:
	//构造函数
	Myset(int a[], int n = 0) : size(n) {
        if (n > 0) {
            ptr = new int[n];
            for (int i = 0; i < n; i++)
                ptr[i] = a[i];           
        } 
		else
            ptr = nullptr;           
    }
	//析构函数
	~Myset(){delete[] ptr;}
	//复制构造函数
	Myset(const Myset& a) : size(a.size) {
	    ptr = new int[size];
	    for (int i = 0; i < size; i++) {
	        ptr[i] = a.ptr[i];
	    }
	}
	//插入一个元素
	void Insert(int x){
		*(ptr+size)=x;
		size++;
	}
	//删除一个元素
	void Delete() {            
	    if (size > 0) {
	        int* newPtr = new int[size - 1];
	        for (int i = 0; i < size - 1; i++) {
	            newPtr[i] = ptr[i];
	        }
	        delete[] ptr;     
	        ptr = newPtr;
	        size--;
	    }
	}
	//算术运算符+
	Myset operator+(const Myset& a) {
	    int minSize = (size < a.size) ? size : a.size;
	    int* newPtr = new int[minSize];
	    for (int i = 0; i < minSize; i++) {
	        newPtr[i] = ptr[i] + a.ptr[i];
	    }
	    return Myset(newPtr, minSize);
	}
	//算术运算符<<
	friend ostream& operator<<(ostream& os,Myset& a){
		for(int i=0;i<a.size;i++){
			os<<*(a.ptr+i)<<" ";
		}
		return os;
	}
	//查找一个元素，返回下标
	int find(int x){
		for(int i=0;i<size;i++){
			if(*(ptr+i)==x){
				return i;
			}	
		}
		return -1; // 如果未找到元素，返回-1
	}
};
```
