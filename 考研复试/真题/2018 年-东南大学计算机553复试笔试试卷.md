#### 1. 输出 1000 以内的完数，格式：完数 = 因子 1 + 因子 2 + … 因子 n。 
例如: 6=1+2+3。
```c++
#include<iostream>
using namespace std;

int main(){
    int ans[1000];
    int idx = 0;
    for(int i=1;i<=1000;i++){
        int sum=0;
        int x=0;
        for(int j=1;j<i;j++){
            if(i%j==0){
                sum+=j;
                ans[x++] = j;
            }
        }
        if(sum==i){
            cout<<i<<" = ";
            for(int k=0;k<x;k++){
                cout<<ans[k];
                if(k!=x-1) cout<<" + ";
            }
            cout<<endl;
        }
    }
}
```

#### 2. 随机生成 10 个到 100000 个数字，转换字符串并输出至文件。 
```c++
#include<iostream>
#include<fstream>
#include<cstdlib>
#include<string>
using namespace std;

int main(){
    srand(time(0));
    int n = rand() % 99991 +10; // 生成一个10到100000之间的随机整数
    ofstream outFile("output.txt");
    for(int i = 0; i < n; i++){
        int num = rand(); // 生成一个随机整数
        string str = to_string(num); 
        outFile << str << endl; 
    }
    outFile.close();
    return 0;
}
```
