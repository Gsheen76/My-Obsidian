#### 1. 给一个三行四列的矩阵按优先随机赋值（1～9 的范围），寻找矩阵的“鞍点” （即行最大值、列最小值）。
```c++
#include<iostream>
#include<ctime>
using namespace std;

int main()
{
    srand(time(0));
    int arr[3][4];

    // 生成1~9随机数
    for(int i=0;i<3;i++){
        for(int j=0;j<4;j++){
            arr[i][j] = rand() % 9 + 1;
        }
    }

    for(int i=0;i<3;i++){
        for(int j=0;j<4;j++){
            cout << arr[i][j] << " ";
        }
        cout << endl;
    }

    // 找鞍点
    bool found = false;

    for(int i=0;i<3;i++){
        for(int j=0;j<4;j++){
            int val = arr[i][j];
            // 判断是否是该行最大
            bool isRowMax = true;
            for(int k=0;k<4;k++){
                if(arr[i][k] > val){
                    isRowMax = false;
                    break;
                }
            }
            // 判断是否是该列最小
            bool isColMin = true;
            for(int k=0;k<3;k++){
                if(arr[k][j] < val){
                    isColMin = false;
                    break;
                }
            }
            // 如果同时满足
            if(isRowMax && isColMin){
                cout << val 
                     << " (" << i << "," << j << ")" << endl;
                found = true;
            }
        }
    }
    if(!found){
        cout << "None" << endl;
    }

    return 0;
}
```
