### 3. 写出下面程序输出结果（考察构造函数、析构函数调用问题）

```c++
using namespace std;

    ~one() { cout << "one destruct" << endl; }

    two(const one&) { cout << "two 1construct" << endl; }
    two(const two&) { cout << "two 2construct" << endl; }
    ~two() { cout << "two destruct" << endl; }
};
void fun(two) {
    cout << "fun()" << endl;
}

int main() {
    one One;
    fun(One);

}
```

