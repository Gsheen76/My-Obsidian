# 二、 读程序 

#### 1、 题目给出如下代码，请说明它的功能，若输入 This is a test，分析对给定输入的输出结果： 
```c++
#include <iostream> 
#include <iomanip> 
#include <string> 
using namespace std; 
 
int main() { 
    string s; 
    getline(cin, s);    
    string word; 
    cout << setw(6) << "word" 
         << setw(8) << "length" << endl; 
    while (!s.empty()) { 
        if (s.find(" ") == s.npos) { 
            word = s.substr(0); 
            cout << setw(6) << word 
                 << setw(8) << word.length() << endl; 
            break; 
        } 
 
        word = s.substr(0, s.find(" ")); 
        cout << setw(6) << word 
             << setw(8) << word.length() << endl; 
        s.erase(0, s.find(" ") + 1); 
    } 
    return 0; 
}
```
