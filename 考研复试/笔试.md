### 3. 汉诺塔

```c++
void hanoi(int n,char A,char B,char C){
	if(n==1) 
		move(A,C);
	else{
		hanoi(n-1,A,C,B);
		move(A,C);
		hanoi(n-1,B,A,C);
	}
}
```

### 4. 从n个人中选k个的组合数问题($C_n^k$)

```c++
int comm(int n,int k){
	if(k>n) 
		return 0;
	else if(n==k||k==0)
		return 1;
	else
		return comm(n-1,k)+comm(n-1,k-1);
}
```

### 9. 虚函数的隐式和显式覆盖

```c++
class Base {
public:
    // 基类必须有 virtual！
    virtual void eat(string food) {
        cout << "Base: 吃" << food << endl;
    }
    
    virtual void sleep(int hours) {
        cout << "Base: 睡觉" << hours << "小时" << endl;
    }
};

// 派生类1：隐式覆盖
class Dog : public Base {
public:
    // 隐式覆盖：不写 override，但函数签名完全一致
    // 注意：这里可以不写 virtual，但基类必须有
    void eat(string food) {  
        cout << "Dog: 开心地吃" << food << endl;
    }
    
    void sleep(int hours) {
        cout << "Dog: 睡" << hours << "小时，看门" << endl;
    }
};

// 派生类2：显式覆盖
class Cat : public Base {
public:
    // 显式覆盖：使用 override 关键字
    void eat(string food) override {
        cout << "Cat: 优雅地吃" << food << endl;
    }
    
    void sleep(int hours) override {
        cout << "Cat: 睡" << hours << "小时，晒太阳" << endl;
    }
};

// 派生类3：错误的实现（参数不一致）
class Bird : public Base {
public:
    // ❌ 这不是覆盖！参数类型不同
    void eat(int food) {  // int vs string
        cout << "Bird: 啄食" << food << "粒米" << endl;
    }
    
    // 如果试图加 override
    // void eat(int food) override {  // ❌ 编译错误！
    //     cout << "Bird: 啄食" << food << "粒米" << endl;
    // }
};
```

### 11. strtok用法

`strtok` 是 C 语言中用于**分割字符串**的函数，它会将字符串按照指定的分隔符拆分成多个子串（标记/token）。
```c++
#include <iostream>
#include <cstring>

int main() {
    char sentence[] = "Hello,world!C++ programming";
    char *token;
    
    // 第一次调用：传入要分割的字符串
    token = strtok(sentence, ",! ");  // 分隔符：逗号、感叹号、空格
    
    // 后续调用：传入 NULL
    while (token != NULL) {
        std::cout << "单词: " << token << std::endl;
        token = strtok(NULL, ",! ");  // 继续获取下一个单词
    }
    
    return 0;
}
```

### 12. setw用法

`setw` 是 C++ 中用于**设置输出字段宽度**的流操纵符（stream manipulator）。
